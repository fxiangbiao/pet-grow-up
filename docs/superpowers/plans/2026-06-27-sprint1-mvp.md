### Task 6: 创建 SceneMathTen.svelte（凑十法场景组件）

**Files:**
- Create: `frontend/src/lib/components/study/SceneMathTen.svelte`

**目标：** 实现"拖拽苹果到碗凑十"的完整场景化关卡。宠物在场景上方提示，苹果散落在屏幕中，孩子拖拽苹果到碗里凑满 10 个。

- [ ] **Step 1: 创建组件文件并写 props 和状态**

```svelte
<script lang="ts">
  import type { QuestionDTO, AnswerResultDTO } from '$lib/types/api';
  import { studyApi } from '$lib/api/study';

  let {
    question,
    sessionId,
    onComplete
  }: {
    question: QuestionDTO;
    sessionId: number;
    onComplete: (result: AnswerResultDTO) => void;
  } = $props();

  // 从题目中解析目标数字（如 "8 + ? = 10" → target=10, start=8）
  const target = 10; // 凑十法固定目标
  const start = parseInt(question.questionText.match(/(\d+)\s*\+/)?.[1] || '0');
  const needed = target - start;

  // 场景状态
  let apples = $state<Array<{ id: number; x: number; y: number; inBowl: boolean }>>([]);
  let bowlCount = $state(0);
  let dragging = $state<number | null>(null);
  let dragX = $state(0);
  let dragY = $state(0);
  let feedback = $state<'idle' | 'correct' | 'tooMany'>('idle');
  let showFeedback = $state(false);

  // 初始化苹果（随机散落 10 个苹果）
  function initApples() {
    apples = Array.from({ length: target }, (_, i) => ({
      id: i,
      x: 30 + Math.random() * 70,  // 百分比位置
      y: 20 + Math.random() * 50,
      inBowl: false
    }));
    bowlCount = 0;
    feedback = 'idle';
    showFeedback = false;
  }
  initApples();

  // 拖拽处理
  function handlePointerDown(id: number, e: PointerEvent) {
    dragging = id;
    dragX = e.clientX;
    dragY = e.clientY;
    (e.target as HTMLElement)?.setPointerCapture?.(e.pointerId);
  }

  function handlePointerMove(e: PointerEvent) {
    if (dragging === null) return;
    dragX = e.clientX;
    dragY = e.clientY;
  }

  function handlePointerUp() {
    if (dragging === null) return;
    // 检测是否落入碗区域（碗在屏幕底部中央）
    const bowlEl = document.getElementById('bowl-zone');
    if (bowlEl) {
      const rect = bowlEl.getBoundingClientRect();
      if (dragX > rect.left && dragX < rect.right && dragY > rect.top && dragY < rect.bottom) {
        const apple = apples.find(a => a.id === dragging);
        if (apple && !apple.inBowl) {
          apple.inBowl = true;
          bowlCount++;
          if (bowlCount === needed) {
            // 刚好凑够 needed 个 = 正确答案
            feedback = 'correct';
            showFeedback = true;
            submitAnswer();
          } else if (bowlCount > needed) {
            feedback = 'tooMany';
            showFeedback = true;
            // 放多了，自动减少一个
            bowlCount--;
            apple.inBowl = false;
            setTimeout(() => { showFeedback = false; feedback = 'idle'; }, 1000);
          }
        }
      }
    }
    dragging = null;
  }

  async function submitAnswer() {
    try {
      const result = await studyApi.submitAnswer(sessionId, {
        questionId: question.id,
        answer: String(needed) // 正确答案是 needed
      });
      setTimeout(() => { onComplete(result); }, 1200); // 留给庆祝动画时间
    } catch (err) {
      console.error('submit failed', err);
    }
  }
</script>

<div class="relative w-full h-full min-h-[480px] bg-gradient-to-b from-amber-50 to-orange-100 overflow-hidden select-none"
  onpointermove={handlePointerMove} onpointerup={handlePointerUp}>

  <!-- 宠物提示区 -->
  <div class="absolute top-4 left-1/2 -translate-x-1/2 text-center z-10">
    <p class="text-lg font-bold text-amber-800 bg-white/70 rounded-full px-6 py-2 shadow-sm">
      🐱「帮我凑 {needed} 个苹果到碗里！」
    </p>
    <p class="text-sm text-amber-600 mt-1">碗里：{bowlCount}/{needed}</p>
  </div>

  <!-- 苹果区 -->
  {#each apples.filter(a => !a.inBowl) as apple (apple.id)}
    <div
      class="absolute w-12 h-12 flex items-center justify-center text-3xl cursor-grab active:cursor-grabbing
        {dragging === apple.id ? 'scale-125 z-20' : 'hover:scale-110'} transition-transform"
      style="left: {apple.x}%; top: {apple.y}%;"
      onpointerdown={(e) => handlePointerDown(apple.id, e)}
      role="button" tabindex="0"
    >
      🍎
    </div>
  {/each}

  <!-- 拖拽跟随的苹果 -->
  {#if dragging !== null && !apples.find(a => a.id === dragging)?.inBowl}
    <div class="absolute w-14 h-14 flex items-center justify-center text-4xl z-30 pointer-events-none"
      style="left: {dragX - 28}px; top: {dragY - 28}px;">
      🍎
    </div>
  {/if}

  <!-- 碗区域 -->
  <div id="bowl-zone"
    class="absolute bottom-8 left-1/2 -translate-x-1/2 w-40 h-28 flex flex-col items-center justify-end
      border-4 border-dashed rounded-b-[80px] transition-all duration-300"
    class:border-green-400={bowlCount === needed}
    class:border-amber-400={bowlCount > 0 && bowlCount < needed}
    class:border-gray-300={bowlCount === 0}
    class:bg-green-100/50={bowlCount === needed}
    class:bg-amber-50/50={bowlCount > 0 && bowlCount < needed}>

    <span class="text-5xl mb-1">🥣</span>
    <span class="text-xs text-gray-500 mb-1">拖苹果到这里</span>
  </div>

  <!-- 反馈提示 -->
  {#if showFeedback}
    <div class="absolute inset-0 flex items-center justify-center z-20 pointer-events-none">
      {#if feedback === 'correct'}
        <div class="bg-green-500 text-white text-2xl font-bold px-8 py-4 rounded-2xl shadow-xl animate-bounce-in">
          ✅ 凑满 {target} 啦！
        </div>
      {:else if feedback === 'tooMany'}
        <div class="bg-amber-500 text-white text-xl font-bold px-6 py-3 rounded-2xl shadow-xl animate-shake">
          ⚠️ 太多啦！只要 {needed} 个～
        </div>
      {/if}
    </div>
  {/if}

  <!-- 碗中苹果计数显示 -->
  <div class="absolute bottom-32 left-1/2 -translate-x-1/2 flex gap-1">
    {#each Array(bowlCount) as _}
      <span class="text-xl">🍎</span>
    {/each}
  </div>
</div>

<style>
  @keyframes bounceIn {
    0% { transform: scale(0.3); opacity: 0; }
    50% { transform: scale(1.1); }
    70% { transform: scale(0.9); }
    100% { transform: scale(1); opacity: 1; }
  }
  :global(.animate-bounce-in) {
    animation: bounceIn 0.5s ease-out;
  }
</style>
```

### Task 9: 创建 Godot API 客户端脚本

**目标：** 封装对 Spring Boot 后端的 HTTP 调用，处理 JWT 认证。

- [ ] **Step 1: 创建 HTTP API 客户端**

```gdscript
# scripts/api_client.gd
# HTTP client for Spring Boot backend (localhost:8080)
extends Node

const BASE_URL = "http://localhost:8080/api/v1"
var access_token: String = ""
var refresh_token: String = ""

signal login_success
signal login_failed(reason: String)

# ── Auth ──
func login(username: String, password: String) -> void:
	var http = HTTPRequest.new()
	add_child(http)
	http.request_completed.connect(_on_login_complete.bind(http))
	
	var body = JSON.stringify({"username": username, "password": password})
	var headers = ["Content-Type: application/json"]
	http.request(BASE_URL + "/auth/login", headers, HTTPClient.METHOD_POST, body)

func _on_login_complete(result: int, code: int, headers: PackedStringArray, body: PackedByteArray, http: HTTPRequest) -> void:
	http.queue_free()
	if code == 200:
		var json = JSON.parse_string(body.get_string_from_utf8())
		access_token = json.get("data", {}).get("accessToken", "")
		refresh_token = json.get("data", {}).get("refreshToken", "")
		login_success.emit()
	else:
		login_failed.emit("Login failed: HTTP " + str(code))

# ── Generic request ──
func request(method: int, path: String, data: Dictionary = {}, callback: Callable = Callable()) -> void:
	var http = HTTPRequest.new()
	add_child(http)
	if callback.is_valid():
		http.request_completed.connect(_on_request_complete.bind(http, callback))
	else:
		http.request_completed.connect(_on_request_log.bind(http))
	
	var url = BASE_URL + path
	var headers = ["Content-Type: application/json"]
	if access_token != "":
		headers.append("Authorization: Bearer " + access_token)
	
	var body = JSON.stringify(data) if not data.is_empty() else ""
	http.request(url, headers, method, body)

func _on_request_complete(result: int, code: int, headers: PackedStringArray, body: PackedByteArray, http: HTTPRequest, cb: Callable) -> void:
	http.queue_free()
	var json = JSON.parse_string(body.get_string_from_utf8())
	cb.call(code, json)

func _on_request_log(result: int, code: int, headers: PackedStringArray, body: PackedByteArray, http: HTTPRequest) -> void:
	http.queue_free()
	print("[API] HTTP %d: %s" % [code, body.get_string_from_utf8()])

# ── Convenience methods ──
func start_session(subject: String, node_key: String, callback: Callable) -> void:
	request(HTTPClient.METHOD_POST, "/quiz/session/start", {
		"subject": subject, "nodeKey": node_key
	}, callback)

func submit_answer(session_id: int, question_id: int, answer: String, callback: Callable) -> void:
	request(HTTPClient.METHOD_POST, "/quiz/session/%d/answer" % session_id, {
		"questionId": question_id, "answer": answer
	}, callback)

func get_spirit(callback: Callable) -> void:
	request(HTTPClient.METHOD_GET, "/spirit/my", {}, callback)
```

	await get_tree().create_timer(2.5).timeout
	bubble.visible = false

func _on_click():

	tween = create_tween()

**Files:**

**目标：** 实现拖拽苹果到碗凑十的完整玩法，对应路径 A 中 SceneMathTen.svelte 的同一场景。

- [ ] **Step 1: 创建关卡逻辑脚本**

```gdscript
# scripts/ten_make_level.gd
extends Node2D

const TARGET = 10
var start_number: int = 8  # from question
var needed: int = TARGET - start_number
var bowl_count: int = 0

@onready var apples_container = $Apples
@onready var bowl = $Bowl

signal level_complete(correct: bool)

func _ready():
	needed = TARGET - start_number
	spawn_apples(TARGET)
	update_hint()

func spawn_apples(count: int):
	for i in range(count):
		var apple = Sprite2D.new()
		apple.texture = preload("res://assets/sprites/apple.png")  # 后续替换为实际美术
		apple.position = Vector2(randf_range(80, 400), randf_range(100, 350))
		apple.scale = Vector2(0.4, 0.4)
		apple.name = "Apple_%d" % i
		
		# Make draggable
		var area = Area2D.new()
		var col = CollisionShape2D.new()
		col.shape = CircleShape2D.new()
		col.shape.radius = 30
		area.add_child(col)
		apple.add_child(area)
		
		area.input_pickable = true
		area.input_event.connect(_on_apple_input.bind(apple))
		
		apples_container.add_child(apple)

func _on_apple_input(viewport: Node, event: InputEvent, shape_idx: int, apple: Sprite2D):
	if event is InputEventMouseButton:
		if event.button_index == MOUSE_BUTTON_LEFT and event.pressed:
			apple.z_index = 10
			_start_drag(apple)
		elif event.button_index == MOUSE_BUTTON_LEFT and not event.pressed:
			apple.z_index = 0
			_end_drag(apple)

func _start_drag(apple: Sprite2D):
	# Follow mouse until release
	set_process_input(true)
	apple.set_meta("dragging", true)

func _input(event: InputEvent):
	if event is InputEventMouseMotion:
		for apple in apples_container.get_children():
			if apple.get_meta("dragging", false):
				apple.position = event.position

func _end_drag(apple: Sprite2D):
	apple.set_meta("dragging", false)
	# Check if dropped into bowl area
	var bowl_rect = Rect2(bowl.position - Vector2(100, 60), Vector2(200, 120))
	if bowl_rect.has_point(apple.position):
		apple.queue_free()
		bowl_count += 1
		update_hint()
		if bowl_count == needed:
			level_complete.emit(true)
		elif bowl_count > needed:
			# Too many — respawn one and show feedback
			bowl_count -= 1
			spawn_apples(1)
			show_too_many_feedback()

func update_hint():
	$HintLabel.text = "拖 %d 个苹果到碗里！碗里：%d/%d" % [needed, bowl_count, needed]

func show_too_many_feedback():
	$TooManyLabel.visible = true
	await get_tree().create_timer(1.0).timeout
	$TooManyLabel.visible = false
```
